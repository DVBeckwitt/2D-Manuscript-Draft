\begin{tikzpicture}[scale=2,>=stealth]
  \def\R{1.2}        % Bragg-sphere radius
  \def\Raxis{1.6}    % axis length
  \def\angleMid{90}  % specular vector, so Q_R is zero
  \def\angStart{60}
  \def\angEnd{90}    % opposite tilt signs fold onto Q_R >= 0

  % Folded Bragg locus in the radial coordinate Q_R >= 0
  \draw[dashed,blue] (0,-\R) arc (-90:90:\R);

  % reciprocal-space axes
  \draw[->,thick] (0,0) -- (0,\Raxis) node[above] {$Q_z$};
  \draw[->,thick] (0,0) -- (\Raxis,0) node[right] {$Q_R$};

  % reciprocal vector for the specular case: purely along Q_z
  \coordinate (Gvec) at ({\R*cos(\angleMid)},{\R*sin(\angleMid)}); % = (0,\R)
  \draw[thick,->] (0,0) -- (Gvec)
       node[pos=0.75, right] {\scriptsize $Q=|\mathbf G_{00l}|$};

  % The two signed tilt branches coincide after folding to radial Q_R.
  \draw[very thick,orange]
    ({\R*cos(\angStart)},{\R*sin(\angStart)}) arc (\angStart:\angEnd:\R)
    node[pos=0.6, above right] {\scriptsize $\omega_{\mathrm w}(\Delta\theta)$};
\end{tikzpicture}
